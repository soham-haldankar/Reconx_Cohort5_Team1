CREATE Table trades (
id BIGSERIAL,
trade_ref VARCHAR(30) NOT NULL,
instrument_id BIGINT,
counterparty_id BIGINT,
asset_class VARCHAR(20) NOT NULL,
side VARCHAR(4),
quantity NUMERIC(18,4),
price NUMERIC(18,4),
trade_date DATE NOT NULL,
status VARCHAR (20) DEFAULT 'PENDING',
deleted_at TIMESTAMPTZ DEFAULT now(),
modified_at TIMESTAMPTZ,
PRIMARY KEY (id, trade_date)
)
PARTITION BY RANGE (trade_date);

CREATE INDEX idx_trades_status
ON trades(status);

CREATE INDEX idx_trades_instrument
ON trades(instrument_id);

CREATE INDEX idx_trades_counterparty
ON trades(counterparty_id);



CREATE TABLE instruments (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY,
    symbol VARCHAR(20) NOT NULL,
    name VARCHAR(200) NOT NULL,
    asset_class VARCHAR(20) NOT NULL,
    currency VARCHAR(3) NOT NULL,
    isin VARCHAR(12),
    CONSTRAINT PK_INSTRUMENTS PRIMARY KEY (id),
    CONSTRAINT UQ_INSTRUMENTS_SYMBOL UNIQUE (symbol),
    CONSTRAINT UQ_INSTRUMENTS_ISIN UNIQUE (isin)
);
